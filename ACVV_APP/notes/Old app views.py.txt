@login_required
def client_information(request):
    records_per_page = 36
    page = request.GET.get('page')
    search_id = request.GET.get('search_id')
    search_first_name = request.GET.get('search_first_name')
    search_surname = request.GET.get('search_surname')

    with connection.cursor() as cursor:
        cursor.execute("SELECT `[No.]`, `ID NUMBER OF DEBTOR`, TOTAL FROM debt_collection_6")
        all_debt_records = cursor.fetchall()

        debtor_ids = [record[1] for record in all_debt_records]

        # Fetch all relevant clients in one go
        clients = ClientList.objects.filter(ID_Number__in=debtor_ids)
        client_dict = {client.ID_Number: client for client in clients}

        filtered_debt_records = []
        for record in all_debt_records:
            debt_no, id_number, total = record
            client = client_dict.get(id_number)
            if client:
                include = True
                if search_id and search_id.lower() not in str(client.ID_Number).lower():
                    include = False
                if search_first_name and search_first_name.lower() not in str(client.FIRST_NAME).lower():
                    include = False
                if search_surname and search_surname.lower() not in str(client.SURNAME).lower():
                    include = False
                if include:
                    filtered_debt_records.append({
                        'debt_no': debt_no,
                        'id_number': id_number,
                        'total_debt': total,
                        'client': client,
                    })

        paginator = Paginator(filtered_debt_records, records_per_page)

        try:
            debt_records_page = paginator.page(page)
        except PageNotAnInteger:
            debt_records_page = paginator.page(1)
        except EmptyPage:
            debt_records_page = paginator.page(paginator.num_pages)

        context = {
            'debt_records_page': debt_records_page,
        }
        return render(request, 'Futura/client_information.html', context)

@login_required
def create_call(request):
    current_datetime = datetime.now().isoformat()
    if request.method == 'POST':
        try:
            agent = request.user.username
            id_number = request.POST.get('id_number')
            phone_number = request.POST.get('phone_number')
            start_time_str = request.POST.get('start_time')
            end_time_str = request.POST.get('end_time')
            direction = request.POST.get('direction')
            outcome = request.POST.get('outcome')
            call_result = request.POST.get('call_result')
            notes = request.POST.get('notes')
            recipient_email = request.POST.get('recipient_email')
            number_of_months = request.POST.get('number_of_months')
            installments = request.POST.get('installments')
            start_month = request.POST.get('start_month')
            day_of_month = request.POST.get('day_of_month')

            start_time = datetime.fromisoformat(start_time_str)
            end_time = datetime.fromisoformat(end_time_str)

            # Retrieve client details for the email subject and body
            client = get_object_or_404(ClientList, ID_Number=id_number)
            
            # Retrieve TOTAL from DebtCollection6 using the correct attribute name 'total_debt'
            debt_record = get_object_or_404(DebtCollection6, id_number=id_number)

            # Define fixed email subject and body with the correct attribute
            subject = "Important Account Reminder"
            body = f"Dear {client.TITLE} {client.SURNAME} {client.FIRST_NAME}, This is a reminder regarding your outstanding account balance of R{debt_record.total_debt}. Please contact us urgently. Thank you, Thipa Attorneys."

            attachment_name = None
            if 'attachment' in request.FILES and request.FILES['attachment']:
                attachment = request.FILES['attachment']
                email = EmailMessage(subject, body, settings.EMAIL_HOST_USER, [recipient_email])
                email.attach(attachment.name, attachment.read(), attachment.content_type)
                email.send()
                attachment_name = attachment.name
            else:
                send_mail(subject, body, settings.EMAIL_HOST_USER, [recipient_email], fail_silently=False)

            call = Call(
                agent=agent,
                id_number=id_number,
                phone_number=phone_number,
                start_time=start_time,
                end_time=end_time,
                direction=direction,
                outcome=outcome,
                call_result=call_result,
                notes=notes,
                recipient_email=recipient_email,
                attachment_name=attachment_name,
                number_of_months=number_of_months,
                installments=installments,
                start_month=start_month,
                day_of_month=day_of_month,
            )
            call.save()
            return redirect('agent_overview')
        except Exception as e:
            print(f"Error processing form submission: {e}")
            return HttpResponse(f"Error: {e}")

    return render(request, 'Futura/create_call_page.html', {'current_datetime': current_datetime})

@login_required
def recon_view(request):
    """
    Handles PDF uploads, client payment submissions, and displays reconciliation data.
    
    This view has been updated to use Django's messages framework for user feedback
    and to optimize data fetching and processing for improved performance.
    """
    if request.method == 'POST':
        # Check if the request is specifically for PDF upload
        if 'pdf_file' in request.FILES and request.POST.get('id_number'):
            id_number = request.POST.get('id_number')
            pdf_file = request.FILES['pdf_file']
            
            try:
                # The .get_or_create() method is a useful Django shortcut that attempts
                # to get an object with the given parameters, creating one if it doesn't exist.
                # This prevents duplicate records if a user re-uploads the same PDF.
                pdf_record, created = PDFRecord.objects.get_or_create(
                    id_number=id_number
                )
                pdf_record.pdf_file.save(pdf_file.name, pdf_file)
                pdf_record.save()

                if created:
                    print(f"Successfully created and saved new PDF for ID: {id_number}")
                    messages.success(request, f"PDF for ID: {id_number} was uploaded successfully.")
                else:
                    print(f"Successfully updated PDF for ID: {id_number}")
                    messages.success(request, f"PDF for ID: {id_number} was updated successfully.")
            except Exception as e:
                print(f"Error saving PDF record: {e}")
                messages.error(request, f"Error saving PDF record: {e}")
            
            return redirect('recon')
        
        # Check if the request is specifically for client payment
        elif 'debtor_no' in request.POST:
            id_number = request.POST.get('debtor_no')
            amount_deposited = request.POST.get('amountDeposited')
            statement_reference = request.POST.get('statementReference')
            date_of_statement = request.POST.get('dateOfStatement')

            try:
                if amount_deposited:
                    amount_deposited = Decimal(amount_deposited)
                else:
                    amount_deposited = Decimal('0.00')

                if date_of_statement:
                    date_of_statement = datetime.strptime(date_of_statement, "%Y-%m-%d").date()
                else:
                    date_of_statement = None
                
                # Create and save the new ClientPayment record.
                ClientPayment.objects.create(
                    id_number=id_number,
                    amount_deposited=amount_deposited,
                    statement_reference=statement_reference,
                    date_of_statement=date_of_statement
                )
                print("Saving ClientPayment:", {
                    'id_number': id_number,
                    'amount_deposited': amount_deposited,
                    'statement_reference': statement_reference,
                    'date_of_statement': date_of_statement,
                })
                messages.success(request, f"Payment for debtor {id_number} was saved successfully.")
            except InvalidOperation:
                messages.error(request, "Error: The amount deposited is not a valid number.")
            except ValueError:
                messages.error(request, "Error: The date of statement is not in a valid format.")
            except Exception as e:
                print("Error saving ClientPayment:", e)
                messages.error(request, f"Error saving client payment: {e}")

            return redirect('recon')


    # --- DATA FETCHING & PROCESSING FOR DISPLAY ---
    clients = ClientList.objects.all()
    client_ids = [client.ID_Number for client in clients]
    
    # Pre-fetch all TraceResult and ClientPayment data with a single query each.
    all_trace_results = TraceResult.objects.filter(ID_Number__in=client_ids)
    payments_data = ClientPayment.objects.filter(id_number__in=client_ids).values(
        'id_number').annotate(total_paid=Sum('amount_deposited'))
    
    # Fetch all PDF records and create a lookup dictionary
    pdf_records = PDFRecord.objects.filter(id_number__in=client_ids)
    pdf_dict = {record.id_number: record for record in pdf_records}

    # Create efficient dictionaries for O(1) lookups.
    trace_results_dict = {}
    for result in all_trace_results:
        if result.ID_Number not in trace_results_dict:
            trace_results_dict[result.ID_Number] = []
        trace_results_dict[result.ID_Number].append(result)

    payments_dict = {item['id_number']: item['total_paid'] for item in payments_data}
    
    # Fetch all debt records for all clients at once using raw SQL for efficiency.
    all_debt_records = {}
    if client_ids:
        # Use a parameterized query to prevent SQL injection.
        with connection.cursor() as cursor:
            # We convert the list to a tuple for the IN clause
            # The %s placeholder expects a single argument, which should be a sequence.
            placeholders = ', '.join(['%s'] * len(client_ids))
            query = f"SELECT `ID NUMBER OF DEBTOR`, `[DEBT NO]`, `TOTAL` FROM debt_collection_6 WHERE `ID NUMBER OF DEBTOR` IN ({placeholders})"
            cursor.execute(query, tuple(client_ids))
            for row in cursor.fetchall():
                id_number, debt_no, total = row
                if id_number not in all_debt_records:
                    all_debt_records[id_number] = []
                all_debt_records[id_number].append({'debt_no': debt_no, 'total': float(total)})
    
    # Calculate the total debt for each client by summing their individual debts.
    debt_totals = {id_num: sum(d['total'] for d in records) for id_num, records in all_debt_records.items()}

    clients_data = []
    for client in clients:
        # Use the pre-built dictionary for an efficient lookup.
        trace_results = trace_results_dict.get(client.ID_Number, [])
        contact_numbers = set()
        for result in trace_results:
            # A more concise way to get all the contact numbers from the object.
            contact_fields = [
                result.cell_number, result.Home1, result.Work1, result.Cell1,
                result.Home2, result.Work2, result.Cell2, result.Home3,
                result.Work3, result.Cell3, result.Home4, result.Work4,
                result.Cell4, result.Home5, result.Work5, result.Cell5,
                result.Home6, result.Work6, result.Cell6, result.Home7,
                result.Work7, result.Cell7, result.Home8, result.Work8,
                result.Cell8, result.Home9, result.Work9, result.Cell9,
                result.Home10, result.Work10, result.Cell10, result.Home11,
                result.Work11, result.Cell11, result.Home12, result.Work12,
                result.Cell12, result.Home13, result.Work13, result.Cell13,
                result.Home14, result.Work14, result.Cell14, result.Home15,
                result.Work15, result.Cell15,
                result.D_O_1, result.D_O_2, result.D_O_3, result.D_O_4, result.D_O_5,
                result.D_O_6, result.D_O_7, result.D_O_8, result.D_O_9, result.D_O_10,
                result.D_O_11, result.D_O_12, result.D_O_13, result.D_O_14, result.D_O_15,
            ]
            contact_numbers.update(num for num in contact_fields if num and num != "NULL")
        contact_details = ", ".join(contact_numbers)
        
        # Get the total debt and total payments from the pre-calculated dictionaries.
        total_debt = debt_totals.get(client.ID_Number, 0)
        total_paid = payments_dict.get(client.ID_Number, Decimal('0'))
        
        # Ensure total_debt is a Decimal before performing subtraction
        try:
            total_debt_decimal = Decimal(str(total_debt))
        except (InvalidOperation, TypeError):
            total_debt_decimal = Decimal('0.00')

        outstanding_balance = total_debt_decimal - total_paid

        # Get the PDF URL if it exists
        pdf_record = pdf_dict.get(client.ID_Number)
        pdf_url = pdf_record.pdf_file.url if pdf_record else None

        clients_data.append({
            'client': client,
            'contact_details': contact_details,
            'total_debt': total_debt,
            'total_paid': total_paid,
            'outstanding_balance': outstanding_balance,
            'bankline_url': f"YOUR_BANKLINE_URL_{client.ID_Number}",
            'debt_records': all_debt_records.get(client.ID_Number, []),
            'pdf_url': pdf_url, # Add the PDF URL to the data
        })

    return render(request, 'recon.html', {'clients_data': clients_data})


@login_required
def view_pdf(request, id_number):
    """
    Retrieves and serves the latest uploaded PDF for a given ID number.
    Returns a 404 error if no PDF is found.
    """
    try:
        # Get the latest PDF record for the given ID number.
        # This will raise a Http404 if not found.
        pdf_record = get_object_or_404(PDFRecord, id_number=id_number)
        
        # Open the file in binary read mode
        with pdf_record.pdf_file.open('rb') as pdf_file:
            response = HttpResponse(pdf_file.read(), content_type='application/pdf')
            # The 'attachment' disposition will force a download, 'inline' will show it in the browser
            response['Content-Disposition'] = f'inline; filename="{pdf_record.pdf_file.name}"'
            return response
    except Http404:
        return HttpResponse("No PDF found for this ID number.", status=404)
    except Exception as e:
        # Catch any other potential errors and return a server error message.
        return HttpResponse(f"An error occurred: {e}", status=500)

def dashboard(request):
    # Get calls per month for each agent in 2025
    calls_data = (
        Call.objects.filter(start_time__year=2025)
        .values("agent", "start_time__month")
        .annotate(call_count=Count("id"))
    )

    # Get total installment value for each agent in 2025
    installment_data = (
        Call.objects.filter(start_time__year=2025)
        .values("agent")
        .annotate(total_installments=Sum("installments"))
    )

    # Prepare data for Chart.js
    calls_by_agent = {}
    for item in calls_data:
        agent = item["agent"]
        month = item["start_time__month"]
        count = item["call_count"]
        if agent not in calls_by_agent:
            calls_by_agent[agent] = [0] * 12  # Initialize with 12 months
        calls_by_agent[agent][month - 1] = count  # Month is 1-based

    installment_values = {}
    for item in installment_data:
        agent = str(item["agent"])  # Convert agent to string
        value = item["total_installments"]
        installment_values[agent] = value

    # Convert calls_by_agent to JSON
    calls_by_agent_json = json.dumps(calls_by_agent)

    context = {
        "calls_by_agent": calls_by_agent,
        "installment_values": installment_values,
        "calls_by_agent_json": calls_by_agent_json,  # Add JSON data to context
    }
    return render(request, "dashboard.html", context)

@login_required
def daily_workflow(request):
    """
    This view manages the daily workflow for a logged-in agent.
    It now filters the workflow based on the client selected from the Agent Dashboard.
    """
    try:
        current_agent = request.user.agent
    except Agent.DoesNotExist:
        messages.error(request, "Agent profile not found.")
        return redirect('login') 

    today = timezone.localdate()

    # Get the 'client' parameter from the URL
    client_param = request.GET.get('client')

    # Start with a base queryset for today's uncompleted debtors for the current agent
    base_queryset = DailyDebtorWorkflow.objects.filter(
        agent=current_agent,
        completion_date=today,
        completed=False
    )
    
    # Filter the queryset based on the client parameter
    if client_param:
        if client_param == 'Limpopo,Kannaland':
            base_queryset = base_queryset.filter(debtor__client_name__in=['Limpopo', 'Kannaland'])
        else:
            base_queryset = base_queryset.filter(debtor__client_name=client_param)

    if request.method == 'POST':
        if 'complete_debtor' in request.POST:
            debtor_no_to_complete = request.POST.get('complete_debtor')
            if debtor_no_to_complete:
                try:
                    with transaction.atomic():
                        workflow_entry = base_queryset.get(
                            debtor__debtor_no=debtor_no_to_complete
                        )
                        workflow_entry.completed = True
                        workflow_entry.save()
                    messages.success(request, f"Debtor {debtor_no_to_complete} marked as completed successfully!")
                except DailyDebtorWorkflow.DoesNotExist:
                    messages.error(request, f"Debtor {debtor_no_to_complete} not found in today's workflow.")
        
        # Correctly build the URL with query parameters
        redirect_url = reverse('daily_workflow')
        if client_param:
            redirect_url += f"?client={client_param}"
        return redirect(redirect_url)

    uncompleted_daily_debtors_queryset = base_queryset
    
    if not uncompleted_daily_debtors_queryset.exists():
        five_days_ago = today - timedelta(days=5)
        
        recently_called_ids = DailyDebtorWorkflow.objects.filter(
            agent=current_agent,
            completed=True,
            completion_date__gte=five_days_ago
        ).values_list('debtor__debtor_no', flat=True)

        new_debtors_qs = DebtCollection6.objects.exclude(debtor_no__in=recently_called_ids)
        if client_param:
            if client_param == 'Limpopo,Kannaland':
                new_debtors_qs = new_debtors_qs.filter(client_name__in=['Limpopo', 'Kannaland'])
            else:
                new_debtors_qs = new_debtors_qs.filter(client_name=client_param)

        debtor_pks_to_add = new_debtors_qs.order_by('?')[:60].values_list('pk', flat=True)
        
        workflow_entries = [
            DailyDebtorWorkflow(
                agent=current_agent,
                debtor_id=pk, 
                completion_date=today
            ) for pk in debtor_pks_to_add
        ]

        DailyDebtorWorkflow.objects.bulk_create(workflow_entries)
    
    daily_debtors_queryset = base_queryset.select_related('debtor')
    
    debtor_id_numbers = daily_debtors_queryset.values_list('debtor__id_number', flat=True).filter(debtor__id_number__isnull=False)

    trace_results_qs = TraceResult.objects.filter(ID_Number__in=debtor_id_numbers)
    
    phone_numbers_dict = {}
    for result in trace_results_qs:
        id_number = result.ID_Number
        if id_number:
            contact_fields = [
                result.cell_number, result.Home1, result.Work1, result.Cell1,
                result.Home2, result.Work2, result.Cell2, result.Home3,
                result.Work3, result.Cell3, result.Home4, result.Work4,
                result.Cell4, result.Home5, result.Work5, result.Cell5,
                result.Home6, result.Work6, result.Cell6, result.Home7,
                result.Work7, result.Cell7, result.Home8, result.Work8,
                result.Cell8, result.Home9, result.Work9, result.Cell9,
                result.Home10, result.Work10, result.Cell10, result.Home11,
                result.Work11, result.Cell11, result.Home12, result.Work12,
                result.Cell12, result.Home13, result.Work13, result.Cell13,
                result.Home14, result.Work14, result.Cell14, result.Home15,
                result.Work15, result.Cell15,
                result.D_O_1, result.D_O_2, result.D_O_3, result.D_O_4, result.D_O_5,
                result.D_O_6, result.D_O_7, result.D_O_8, result.D_O_9, result.D_O_10,
                result.D_O_11, result.D_O_12, result.D_O_13, result.D_O_14, result.D_O_15,
            ]
            
            if id_number not in phone_numbers_dict:
                phone_numbers_dict[id_number] = set()

            for num in contact_fields:
                if num and num != "NULL":
                    phone_numbers_dict[id_number].add(num)
    
    for key in phone_numbers_dict:
        phone_numbers_dict[key] = list(phone_numbers_dict[key])
    
    debtor_id_numbers = daily_debtors_queryset.values_list('debtor__id_number', flat=True).filter(debtor__id_number__isnull=False)
    
    payments_data = ClientPayment.objects.filter(id_number__in=debtor_id_numbers).values(
        'id_number').annotate(total_paid=Sum('amount_deposited'))
    payments_dict = {item['id_number']: item['total_paid'] for item in payments_data}
    
    daily_debtors = []
    for entry in daily_debtors_queryset:
        if entry.debtor.id_number:
            total_debt = entry.debtor.total_debt
            total_paid = payments_dict.get(entry.debtor.id_number, Decimal('0'))
            
            try:
                total_debt_decimal = Decimal(str(total_debt))
            except (InvalidOperation, TypeError):
                total_debt_decimal = Decimal('0.00')

            outstanding_balance = total_debt_decimal - total_paid
            
            daily_debtors.append({
                'debtor_no': entry.debtor.debtor_no,
                'debtor_name': entry.debtor.debtor_name,
                'id_number': entry.debtor.id_number,
                'total_debt': total_debt,
                'outstanding_balance': outstanding_balance,
                'completed': entry.completed,
                'phone_numbers': phone_numbers_dict.get(entry.debtor.id_number, []),
            })
        
    context = {
        'daily_debtors': daily_debtors,
    }
    return render(request, 'Futura/daily_workflow.html', context)

# Function to generate a bar chart image from chart data with labels
def create_payment_chart_image(chart_data):
    """
    Generates a bar chart with data labels on top of each bar, saves it to a buffer,
    and returns the buffer for embedding in a PDF.
    """
    labels = chart_data.get('labels', [])
    data = chart_data.get('data', [])

    # Create a DataFrame for easy plotting
    df = pd.DataFrame({'Month': labels, 'Total Paid': data})
    
    # Use a professional seaborn style
    plt.style.use('seaborn-v0_8-whitegrid')
    plt.figure(figsize=(8, 5))
    
    # Create the bar plot
    ax = sns.barplot(x='Month', y='Total Paid', data=df, color='skyblue')
    
    # Add labels and title
    plt.title('Monthly Payments', fontsize=16, fontweight='bold')
    plt.xlabel('Month', fontsize=12)
    plt.ylabel('Total Paid (R)', fontsize=12)
    
    # Add value labels on top of the bars
    for p in ax.patches:
        height = p.get_height()
        ax.annotate(f'R{height:,.2f}', 
                    (p.get_x() + p.get_width() / 2., height), 
                    ha='center', va='center', 
                    xytext=(0, 9), 
                    textcoords='offset points')
    
    # Save the chart to a buffer
    buf = io.BytesIO()
    plt.savefig(buf, format='png', bbox_inches='tight')
    buf.seek(0)
    plt.close()
    
    return buf

# If modifying these scopes, delete the file token.json.
SCOPES = ['https://www.googleapis.com/auth/gmail.readonly']

def get_gmail_service():
    """
    Handles authentication with the Gmail API.
    Returns the Gmail service object, or a redirect URL if authentication is required.
    """
    creds = None
    if os.path.exists('token.json'):
        creds = Credentials.from_authorized_user_file('token.json', SCOPES)
    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        else:
            flow = InstalledAppFlow.from_client_secrets_file('credentials.json', SCOPES)
            flow.redirect_uri = os.environ.get("NGROK_TUNNEL") + 'futura/inbox/'
            auth_url, _ = flow.authorization_url(prompt='consent', access_type='offline')
            return auth_url
        with open('token.json', 'w') as token:
            token.write(creds.to_json())
    return build('gmail', 'v1', credentials=creds)

def fetch_gmail_inbox(request):
    """
    Fetches a paginated list of emails from the user's inbox.
    Manages pagination using a session to store page tokens.
    """
    print('--- fetch_gmail_inbox view initiated ---')
    service = get_gmail_service()

    # If the service returns a string, it's a redirect URL for authentication
    if isinstance(service, str):
        return redirect(service)

    email_data = []
    page_token = request.GET.get('page_token', None)
    
    # Initialize a list in the session to store page tokens if it doesn't exist
    if 'page_token_history' not in request.session:
        request.session['page_token_history'] = [None]
    
    # Check for 'previous' button action
    if request.GET.get('action') == 'previous':
        if len(request.session['page_token_history']) > 1:
            # Remove the current page token
            request.session['page_token_history'].pop()
            # Get the previous page token from the history
            page_token = request.session['page_token_history'][-1]
    elif page_token:
        # If a new page_token is provided, add it to the history
        if page_token not in request.session['page_token_history']:
            request.session['page_token_history'].append(page_token)
    
    try:
        results = service.users().messages().list(
            userId='me',
            maxResults=12,
            pageToken=page_token
        ).execute()
        
        email_messages = results.get('messages', [])
        new_next_page_token = results.get('nextPageToken', None)

        for msg in email_messages:
            msg_id = msg['id']
            message_data = service.users().messages().get(
                userId='me',
                id=msg_id,
                format='metadata',
                metadataHeaders=['From', 'Subject', 'Date']
            ).execute()
            
            headers = {header['name']: header['value'] for header in message_data.get('payload', {}).get('headers', [])}
            
            email_data.append({
                'id': msg_id,
                'sender': headers.get('From', 'No Sender'),
                'subject': headers.get('Subject', 'No Subject'),
                'received': headers.get('Date', 'No Date'),
                'body': message_data.get('snippet', '')
            })

    except Exception as e:
        print(f"An error occurred: {e}")
        return HttpResponse(f"An error occurred: {e}", status=500)

    # Check if there is a previous page available
    has_previous_page = len(request.session['page_token_history']) > 1
    
    context = {
        'email_data': email_data,
        'next_page_token': new_next_page_token,
        'has_previous_page': has_previous_page,
    }
    return render(request, 'gmail_inbox.html', context)


# The get_email view for a single email remains unchanged and is correctly implemented.
def get_email(request, email_id):
    service = get_gmail_service()
    if isinstance(service, str):
        return JsonResponse({'error': 'Authentication required'}, status=401)
    
    try:
        message = service.users().messages().get(userId='me', id=email_id, format='full').execute()
        
        headers = {header['name']: header['value'] for header in message.get('payload', {}).get('headers', [])}
        from_email = headers.get('From', 'No Sender')
        subject = headers.get('Subject', 'No Subject')
        
        email_body = ""
        payload = message.get('payload', {})
        
        def get_body(parts):
            body_text = ""
            for part in parts:
                if part.get('parts'):
                    body_text += get_body(part['parts'])
                elif part.get('body', {}).get('data'):
                    data = part['body']['data']
                    mime_type = part['mimeType']
                    decoded_data = base64.urlsafe_b64decode(data).decode('utf-8', errors='ignore')
                    
                    if mime_type == 'text/html':
                        return decoded_data
                    if mime_type == 'text/plain' and not body_text:
                        body_text = decoded_data
            return body_text
            
        email_body = get_body([payload])

        response_data = {
            'subject': subject,
            'from_email': from_email,
            'body': email_body,
        }
        
        return JsonResponse(response_data)

    except Exception as e:
        print(f"An error occurred while fetching email: {e}")
        return JsonResponse({'error': f"An error occurred: {e}"}, status=500)
